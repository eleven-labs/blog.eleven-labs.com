import { Meta } from '@storybook/addon-docs';
import { Layout, Table } from '@/storybook/components';
import { Text } from '@/design-system';

import { getValueOfCssPropertyInDesignTokens } from '@/design-system/helpers/storybookHelper';
import { get } from '@/design-system/helpers/objectHelper';

<Meta

                label: 'Name',
            },
            {
                name: 'preview',
                label: 'Preview',
            },
            {

            })} / ${getValueOfCssPropertyInDesignTokens({
              path: 'typography.text',
              tokenName,
              propertyCSS: 'line-height'
            })}`,
            desktop: `${getValueOfCssPropertyInDesignTokens({
                path: 'typography.text',
                tokenName,
                propertyCSS: 'font-size',
                isDesktop: true,
            })} / ${getValueOfCssPropertyInDesignTokens({
              path: 'typography.text',
